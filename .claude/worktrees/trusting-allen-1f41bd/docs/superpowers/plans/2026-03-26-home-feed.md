**Goal:** Build the Home page — personalized evidence feed with filter tabs, paper cards, right sidebar widgets, card more-menu, and paper digest modal.

**Architecture:** Page component at `/` renders a greeting header, filter tabs, and a 2-column card grid sourced from a `useFeed()` hook backed by mock data. Right sidebar shows 3 stat widgets. All data is mock/static with simulated loading delays.

- [ ] **Step 2: Create `useFeed()` hook with loading state**

Create `src/hooks/useFeed.ts` following the spec's `{ data, isLoading, error }` pattern with simulated loading delay:

```ts
import { useState, useMemo, useEffect } from "react";
import { mockPapers, type Paper } from "@/lib/mock-data/papers";
import { useUserStore } from "@/stores/user-store";

export type FeedTab = "for-you" | "trending" | "by-specialty";

export function useFeed() {
  const [activeTab, setActiveTab] = useState<FeedTab>("for-you");
  const [specialtyFilter, setSpecialtyFilter] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [paperActions, setPaperActions] = useState<Record<string, { liked: boolean; saved: boolean; readLater: boolean }>>({});
  const preferences = useUserStore((s) => s.preferences);

  // Simulate initial load delay
  useEffect(() => {
    setIsLoading(true);
    const timer = setTimeout(() => setIsLoading(false), 400);
    return () => clearTimeout(timer);
  }, [activeTab, specialtyFilter]);

  const papers = useMemo(() => {
    const withActions = mockPapers.map((p) => ({
      ...p,
      ...paperActions[p.id],
    }));

    switch (activeTab) {
      case "for-you":
        return [...withActions].sort((a, b) => {
          const aMatch = a.specialty === preferences.specialty ? -1 : 0;
          const bMatch = b.specialty === preferences.specialty ? -1 : 0;
          return aMatch - bMatch;
        });
      case "trending":
        return [...withActions].sort((a, b) => b.evidenceRating - a.evidenceRating);
      case "by-specialty":
        if (!specialtyFilter) return withActions;
        return withActions.filter((p) => p.specialty === specialtyFilter);
      default:
        return withActions;
    }
  }, [activeTab, specialtyFilter, preferences.specialty, paperActions]);

  const specialties = useMemo(
    () => [...new Set(mockPapers.map((p) => p.specialty))],
    []
  );

  function toggleAction(id: string, action: "liked" | "saved" | "readLater") {
    setPaperActions((prev) => ({
      ...prev,
      [id]: {
        liked: prev[id]?.liked ?? mockPapers.find((p) => p.id === id)?.liked ?? false,
        saved: prev[id]?.saved ?? mockPapers.find((p) => p.id === id)?.saved ?? false,
        readLater: prev[id]?.readLater ?? mockPapers.find((p) => p.id === id)?.readLater ?? false,
        [action]: !(prev[id]?.[action] ?? mockPapers.find((p) => p.id === id)?.[action] ?? false),
      },
    }));
  }

  return {
    papers,
    isLoading,
    activeTab,
    setActiveTab,
    specialtyFilter,
    setSpecialtyFilter,
    specialties,
    toggleAction,
  };
}
```
